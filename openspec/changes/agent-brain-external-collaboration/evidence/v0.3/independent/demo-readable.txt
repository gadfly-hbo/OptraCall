function artifactRecords(version=deliveryVersion(),mode='pending'){
  const revision=version==='v1.1'?2:1;
  return ['diff','notes'].map(key=>({key,version,revision,name:(key==='diff'?'member-differences-':'verification-notes-')+version+(key==='diff'?'.csv':'.md'),local:mode,hash:mode==='failed'?'missing':'pending',claimed:mode==='failed'&&key==='diff'?'':key==='diff'?(revision===2?'demo:sha256:6bd9…82e3':'demo:sha256:5ac8…71d2'):(revision===2?'demo:sha256:f0a3…1b29':'demo:sha256:ef92…0a18'),observed:'',bytes:null,reason:mode==='failed'?'未取得文件字节 / 摘要缺失':''}));
}

function currentArtifacts(){
  if(state.artifacts?.version===deliveryVersion())return state.artifacts.items;
  return artifactRecords(deliveryVersion(),state.status==='failed'?'failed':'link-only');
}

function allArtifactsPreserved(){return currentArtifacts().length===2&&currentArtifacts().every(a=>a.local==='preserved'&&a.hash==='matched'&&a.claimed===a.observed)}

function prepareArtifactPreservation(){
  const version=deliveryVersion();
  if(state.artifacts?.version===version)return;
  if(state.artifacts)state.artifactHistory.push(JSON.parse(JSON.stringify(state.artifacts)));
  state.artifacts={version,items:artifactRecords(version)};

  event('附件已登记，等待模拟本机保全',version+' 两件外部产物先登记为待保全；外部链接和清单不代表本机已取得字节。','Host / 模拟附件回执');
  for(const [i,key] of ['diff','notes'].entries())later(()=>{
    if(!activeAuth()||state.artifacts?.version!==version)return;
    const a=state.artifacts.items.find(x=>x.key===key);
    if(!a||a.local!=='pending')return;
    a.local='preserved';a.hash='matched';a.observed=a.claimed;a.bytes=key==='diff'?(version==='v1.1'?4480:4160):(version==='v1.1'?2560:2368);
    event('附件模拟字节保全与摘要核对通过',a.name+'：模拟本机字节 '+a.bytes+' B；外部声明摘要与本机模拟摘要一致。未生成真实下载文件。','Host / 模拟保全回执');
    save();render();
  },650+i*600);
}
